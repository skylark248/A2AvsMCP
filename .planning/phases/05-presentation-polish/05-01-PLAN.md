```typescript
export function GlossaryTerm({ term, children }: GlossaryTermProps) {
  const definition = glossaryTerms[term];
  if (!definition) return <>{children}</>;
  return (
    <Tooltip title={definition} arrow>
      <span
        style={{
          borderBottom: "1px dashed currentColor",
          cursor: "help",
          textDecoration: "none",
        }}
      >
        {children}
      </span>
    </Tooltip>
  );
}
```

Key details per locked decisions:
- D-03: MUI Tooltip on hover, one-sentence definition, no click required
- D-04: Dotted underline via `borderBottom: '1px dashed currentColor'` and `cursor: 'help'`
- Fallback: if term not in map, render `<>{children}</>` unwrapped (no tooltip, no underline)
- Uses inline `style` on span (not `sx` prop since span is not an MUI component). The ProtocolEnvelopeDrawer analog uses `Box component="span" sx={...}` but a plain `<span style={...}>` is simpler and avoids an extra MUI wrapper. Either approach works -- the span is needed because MUI Tooltip requires a DOM ref on its direct child.
