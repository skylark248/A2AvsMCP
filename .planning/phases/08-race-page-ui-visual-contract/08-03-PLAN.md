```typescript
import type { RaceEvent, RaceLane, RaceState, LaneState } from "../../lib/types/race";

const EVENTS_FEED_CAP = 200;

const LANES: RaceLane[] = ["pure_mcp", "pure_a2a", "hybrid"];

function emptyLane(lane: RaceLane): LaneState {
  return {
    lane,
    last_turn_index: -1,
    ttff_ms: null,
    recovered_count: 0,
    total_count: 0,
    faults: [],
    events: [],
    terminal_tag: null,
    headline: null,
  };
}

export const initialRaceState: RaceState = {
  pageState: "pre-race",
  lanes: {
    pure_mcp: emptyLane("pure_mcp"),
    pure_a2a: emptyLane("pure_a2a"),
    hybrid: emptyLane("hybrid"),
  },
  ws_status: "connecting",
  run_id: null,
};

function updateLane(state: RaceState, lane: RaceLane, mutator: (l: LaneState) => LaneState): RaceState {
  return { ...state, lanes: { ...state.lanes, [lane]: mutator(state.lanes[lane]) } };
}

function appendEvent(lane: LaneState, event: RaceEvent): LaneState {
  const events = [...lane.events, event].slice(-EVENTS_FEED_CAP);
  return { ...lane, events };
}

export function raceReducer(state: RaceState, event: RaceEvent): RaceState {
  switch (event.type) {
    case "tick":
    case "tool_call":
    case "agent_msg": {
      return updateLane(state, event.lane, (l) => ({
        ...appendEvent(l, event),
        last_turn_index: Math.max(l.last_turn_index, event.turn_index),
      }));
    }
    case "fault_injected": {
      return updateLane(state, event.lane, (l) => ({
        ...appendEvent(l, event),
        last_turn_index: Math.max(l.last_turn_index, event.turn_index),
        faults: [...l.faults, { fault_id: event.fault_id, fault_kind: event.fault_kind, target: event.target, observed: false }],
      }));
    }
    case "fault_observed": {
      return updateLane(state, event.lane, (l) => {
        const faults = l.faults.map((f) => f.fault_id === event.fault_id ? { ...f, observed: true } : f);
        const isFirstObservation = l.ttff_ms === null;
        return {
          ...appendEvent(l, event),
          last_turn_index: Math.max(l.last_turn_index, event.turn_index),
          faults,
          ttff_ms: isFirstObservation ? event.t_observed_ms : l.ttff_ms,
        };
      });
    }
    case "done": {
      return updateLane(state, event.lane, (l) => ({
        ...appendEvent(l, event),
        last_turn_index: Math.max(l.last_turn_index, event.turn_index),
        terminal_tag: event.tag,
        headline: event.headline,
      }));
    }
    case "error": {
      return updateLane(state, event.lane, (l) => appendEvent(l, event));
    }
    case "race_done": {
      // session-level — no per-lane mutation; consumers read from derivePageState
      return state;
    }
    case "ws_closed":
    case "ws_error": {
      return { ...state, ws_status: "closed" };
    }
    default: {
      const _exhaustive: never = event;
      void _exhaustive;
      return state;
    }
  }
}
```

```typescript
import { useEffect, useReducer, useRef } from "react";
import { initialRaceState, raceReducer } from "../raceReducer";
import type { RaceLane, RaceState } from "../../../lib/types/race";

const WS_PATH = "/api/race/ws";

function buildWsUrl(state: RaceState): string {
  const lanes: RaceLane[] = ["pure_mcp", "pure_a2a", "hybrid"];
  const qs = lanes
    .map((l) => `${l}=${state.lanes[l].last_turn_index}`)
    .join("&");
  // Use page origin; ws-or-wss based on protocol.
  const proto = window.location.protocol === "https:" ? "wss:" : "ws:";
  return `${proto}//${window.location.host}${WS_PATH}?${qs}`;
}

// `enabled` (default true): when false, skip WebSocket open + cleanup. Used by Plan 06
// mobile-summary branch to avoid mounting the ws connection before its early return.
// React rule of hooks forbids conditional hook calls — the consumer must call this hook
// unconditionally and pass `enabled=false` instead.
export function useRaceStream(enabled: boolean = true): RaceState {
  const [state, dispatch] = useReducer(raceReducer, initialRaceState);
  const stateRef = useRef(state);
  stateRef.current = state;

  useEffect(() => {
    if (!enabled) return;                            // gated: Plan 06 mobile branch passes false
    let active = true;
    const url = buildWsUrl(stateRef.current);
    const socket = new WebSocket(url);

    socket.onmessage = (ev) => {
      if (!active) return;
      try {
        const parsed = JSON.parse(ev.data);
        dispatch(parsed);
      } catch {
        // Drop malformed payloads silently (T-08-XX mitigation)
      }
    };
    socket.onerror = () => { if (active) dispatch({ type: "ws_error" }); };
    socket.onclose = () => { if (active) dispatch({ type: "ws_closed" }); };

    return () => {
      active = false;
      socket.close();
    };
  }, [enabled]); // single connection per enabled-state; reducer holds cursor across messages

  return state;
}
```

```typescript
import { useEffect, useState } from "react";
import { fetchRaceReplay, isValidRunId, type RaceReplayPayload } from "../../../lib/api/client";

interface UseRaceReplayResult {
  trace: RaceReplayPayload | null;
  loading: boolean;
  error: string | null;
}

export function useRaceReplay(run_id: string | undefined): UseRaceReplayResult {
  const [trace, setTrace] = useState<RaceReplayPayload | null>(null);
  const [loading, setLoading] = useState<boolean>(Boolean(run_id));
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!run_id) {
      setLoading(false);
      setTrace(null);
      setError(null);
      return;
    }
    if (!isValidRunId(run_id)) {
      setLoading(false);
      setTrace(null);
      setError("Invalid run_id");
      return;
    }

    let active = true;
    setLoading(true);
    setError(null);

    fetchRaceReplay(run_id)
      .then((payload) => {
        if (active) {
          setTrace(payload);
          setError(null);
        }
      })
      .catch((err: unknown) => {
        if (active) {
          setTrace(null);
          setError(err instanceof Error ? err.message : "Failed to load replay.");
        }
      })
      .finally(() => {
        if (active) setLoading(false);
      });

    return () => { active = false; };
  }, [run_id]);

  return { trace, loading, error };
}
```
