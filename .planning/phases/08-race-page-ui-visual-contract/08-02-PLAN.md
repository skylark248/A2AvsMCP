Step 1 — Create `frontend/src/features/race/pageState.ts` exporting `derivePageState(state: { ws_status; lanes; run_id }): PageState`. The derivation function must cover all 12 UIRACE-02 states. Use UI-SPEC Page State Matrix (lines 286-302) as the truth table:

```typescript
import type { PageState, RaceLane, LaneState } from "../../lib/types/race";

export interface DerivePageStateInput {
  ws_status: "connecting" | "open" | "reconnecting" | "closed";
  lanes: Record<RaceLane, LaneState>;
  run_id: string | null;
  countdown_seconds?: number | null; // optional pre-race countdown ticker
  expected_n: number; // 1 or 5
  heatmap_has_data: boolean;
}

export function derivePageState(input: DerivePageStateInput): PageState {
  // Replay mode dominates per D-48: any run_id forces "replay".
  if (input.run_id) return "replay";

  // WS lifecycle states preempt content states.
  if (input.ws_status === "closed") return "ws-disconnected";
  if (input.ws_status === "reconnecting") return "ws-reconnecting";

  // Lane-failed: any lane reached "error" terminal.
  const laneStates = Object.values(input.lanes);
  if (laneStates.some((l) => l.terminal_tag === null && l.events.some((e) => e.type === "error"))) {
    return "lane-failed";
  }

  // Indeterminate: every lane terminal AND at least one terminal_tag === "indeterminate".
  const allTerminal = laneStates.every((l) => l.terminal_tag !== null);
  if (allTerminal) {
    if (laneStates.some((l) => l.terminal_tag === "indeterminate")) return "indeterminate";
    if (!input.heatmap_has_data) return "heatmap-empty";
    // sparse-heatmap: terminal but partial cell coverage. Plan 05 wires the heuristic.
    return "done";
  }

  // Pre-race / countdown / live.
  const anyEvents = laneStates.some((l) => l.events.length > 0);
  if (!anyEvents) {
    if (input.countdown_seconds !== null && input.countdown_seconds !== undefined) return "countdown";
    return "pre-race";
  }
  return input.expected_n >= 5 ? "live-n5" : "live-n1";
}
```
