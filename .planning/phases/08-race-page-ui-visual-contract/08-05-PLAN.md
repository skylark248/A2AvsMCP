```tsx
import { Box, Stack, Typography } from "@mui/material";
import { visuallyHidden } from "@mui/utils";   // canonical accessible-hide helper (NOT srOnly)
import { failureTagColor } from "../../../lib/trace/eventColors";
import type { FailureTag, RaceLane } from "../../../lib/types/race";

export type HardnessType = "long_chain" | "rate_pressure" | "schema_variance" | "multi_source_synthesis";

const HARDNESS_ORDER: HardnessType[] = ["long_chain", "rate_pressure", "schema_variance", "multi_source_synthesis"];
const HARDNESS_LABEL: Record<HardnessType, string> = {
  long_chain: "Long Chain",
  rate_pressure: "Rate Pressure",
  schema_variance: "Schema Variance",
  multi_source_synthesis: "Multi-Source Synthesis",
};

const LANE_ORDER: RaceLane[] = ["pure_mcp", "pure_a2a", "hybrid"];
const LANE_LABEL: Record<RaceLane, string> = {
  pure_mcp: "Pure MCP",
  pure_a2a: "Pure A2A",
  hybrid: "Hybrid",
};

export interface HeatmapCell {
  tag: FailureTag;
  recoveryFraction: string;     // "12/15" — the visible primary text channel
}

export type HeatmapCells = Partial<Record<HardnessType, Partial<Record<RaceLane, HeatmapCell>>>>;

export function HeatmapScaffold({ cells }: { cells: HeatmapCells }) {
  const isEmpty = Object.keys(cells).length === 0
    || HARDNESS_ORDER.every((h) => !cells[h] || Object.keys(cells[h]!).length === 0);

  return (
    <Box sx={{ position: "relative" }} data-testid="heatmap-scaffold">
      <Box
        role="grid"
        aria-label="Hardness vs Failure Heatmap"
        sx={{
          display: "grid",
          gridTemplateColumns: `auto repeat(${LANE_ORDER.length}, 1fr)`,
          gridTemplateRows: `auto repeat(${HARDNESS_ORDER.length}, minmax(44px, 1fr))`,  // 44px touch target
          gap: 0,
          border: "1px solid rgba(16, 32, 51, 0.08)",
        }}
      >
        {/* Header row: empty corner + 3 lane labels */}
        <Box role="columnheader" aria-hidden="true" />
        {LANE_ORDER.map((lane) => (
          <Box key={lane} role="columnheader" sx={{ p: 1, fontWeight: 700, textAlign: "center" }}>
            {LANE_LABEL[lane]}
          </Box>
        ))}

        {/* Body rows */}
        {HARDNESS_ORDER.map((row) => (
          <Box key={`row-${row}`} sx={{ display: "contents" }}>
            <Box role="rowheader" sx={{ p: 1, fontWeight: 700 }}>{HARDNESS_LABEL[row]}</Box>
            {LANE_ORDER.map((lane) => {
              const cell = cells[row]?.[lane];
              const cfg = cell ? failureTagColor[cell.tag] : null;
              const Icon = cfg?.Icon;
              return (
                <Box
                  key={`cell-${row}-${lane}`}
                  role="gridcell"
                  tabIndex={0}
                  data-testid={`heatmap-cell-${row}-${lane}`}
                  sx={{
                    bgcolor: cfg ? cfg.bg : "action.hover",     // CHANNEL 1 (color); neutral muted in empty (D-47)
                    color: cfg ? cfg.text : "text.disabled",
                    borderRadius: 0,                             // UIRACE-03 cell=0
                    p: 1,
                    minHeight: 44,                               // UIRACE-06 touch target
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                    "&:focus-visible": {
                      outline: "3px solid #17475f",              // UI-SPEC focus-visible
                      outlineOffset: 2,
                      "@media (prefers-contrast: more)": {
                        outline: "4px solid #17475f",            // UI-SPEC high-contrast widen
                      },
                    },
                  }}
                >
                  {cell && Icon ? (
                    <Stack direction="row" spacing={0.5} alignItems="center">
                      {/* CHANNEL 2: icon, visibly rendered */}
                      <Box component="span" data-testid="heatmap-cell-icon" aria-hidden="true">
                        <Icon fontSize="small" />
                      </Box>
                      {/* CHANNEL 3: visible recovery-fraction (primary text channel) */}
                      <Typography variant="caption" sx={{ fontWeight: 600 }}>{cell.recoveryFraction}</Typography>
                      {/* CHANNEL 4: sr-only tag-name label via @mui/utils visuallyHidden — NOT srOnly */}
                      <Box component="span" sx={visuallyHidden as object}>{cfg!.label}</Box>
                    </Stack>
                  ) : null}
                </Box>
              );
            })}
          </Box>
        ))}
      </Box>

      {/* heatmap-empty overlay (D-47): grid stays mounted; absolute-positioned center copy */}
      {isEmpty ? (
        <Box
          data-testid="heatmap-empty-overlay"
          sx={{
            position: "absolute", inset: 0,
            display: "flex", flexDirection: "column",
            alignItems: "center", justifyContent: "center",
            pointerEvents: "none",
            bgcolor: "rgba(243, 239, 231, 0.85)",   // background.default semi
          }}
        >
          <Typography variant="h3" sx={{ color: "primary.main" }}>No runs yet</Typography>
          <Typography variant="body2" sx={{ color: "text.secondary" }}>Launch a race to populate the heatmap.</Typography>
        </Box>
      ) : null}
    </Box>
  );
}
```

```tsx
import { Box, Slider, Stack, Typography } from "@mui/material";
import { useEffect, useRef, useState } from "react";

const ANNOUNCE_THROTTLE_MS = 200;

export function ReplayScrubber({
  value,
  max,
  onScrub,
}: {
  value: number;
  max: number;
  onScrub: (turnIndex: number) => void;
}) {
  // Live announcement text — throttled during drag, full on release.
  const [announceText, setAnnounceText] = useState<string>(`Turn ${value} of ${max}`);
  const lastAnnounceRef = useRef<number>(0);
  const pendingValueRef = useRef<number | null>(null);

  // Initial render keeps announce in sync with value.
  useEffect(() => {
    setAnnounceText(`Turn ${value} of ${max}`);
  }, [value, max]);

  const handleChange = (_evt: Event, raw: number | number[]) => {
    const next = Array.isArray(raw) ? raw[0] : raw;
    onScrub(next);

    const now = Date.now();
    pendingValueRef.current = next;
    if (now - lastAnnounceRef.current >= ANNOUNCE_THROTTLE_MS) {
      setAnnounceText(`Turn ${next} of ${max}`);
      lastAnnounceRef.current = now;
      pendingValueRef.current = null;
    }
  };

  // onChangeCommitted = drag release; flush the pending value (D-49 + 08-CONTEXT specifics line 117)
  const handleChangeCommitted = (_evt: Event | React.SyntheticEvent, raw: number | number[]) => {
    const next = Array.isArray(raw) ? raw[0] : raw;
    setAnnounceText(`Turn ${next} of ${max}`);
    lastAnnounceRef.current = Date.now();
    pendingValueRef.current = null;
  };

  return (
    <Stack spacing={1} sx={{ px: 2, py: 1 }}>
      <Box sx={{ minHeight: 40 }} data-testid="race-scrubber">       {/* UI-SPEC line 50 */}
        <Slider
          value={value}
          min={0}
          max={max}
          step={1}
          onChange={handleChange}
          onChangeCommitted={handleChangeCommitted}
          aria-label="Replay turn scrubber"
          sx={{
            "@media (prefers-reduced-motion: reduce)": {
              transitionDuration: "0ms !important",                  // UI-SPEC line 128
              "& .MuiSlider-thumb, & .MuiSlider-track": { transitionDuration: "0ms !important" },
            },
          }}
        />
      </Box>
      <Box aria-live="polite" data-testid="race-scrubber-announce">
        <Typography variant="caption">{announceText}</Typography>
      </Box>
    </Stack>
  );
}
```
