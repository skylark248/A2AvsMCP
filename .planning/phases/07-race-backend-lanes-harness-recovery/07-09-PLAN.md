```
    task_spec: TaskSpec,
    run_id: str,
    recorder: TraceRecorder,

    #    a. LLM call (sonnet_client.messages.create) -> emit llm_call event

    #    e. Tool call / agent message via transport
    #    f. Score gate met or step exhausted
    # 4. Compute ScoreCard via task module's score()
    # 5. Finalize Detectors at done with score_pass
    # 6. Reset ContextVars (always, in finally)
    # 7. Return RaceResult
```

```
Calls race MCP servers via MCPClient(transport='in_process'). Faults flow
through the mock chokepoint per D-25. Detector wiring per D-32 — one Detector
per fault_injected event observed in recorder; consume() on each subsequent
event; record fault_observed with compute_wasted_tokens() on transition.
"""
from __future__ import annotations

import time
import uuid
from typing import Any

from ...mcp.client import MCPClient
from ...mcp_servers.race_context import set_mcp_tool_context, MCP_TOOL_CONTEXT
from ...trace import TraceRecorder
from ..classifier import Detector
from ..failure import InjectedFaultError, FailureScriptEntry
from ..metrics import compute_wasted_tokens
from ..mocks import ActiveFault, ACTIVE_FAULTS, set_active_faults
from ..tasks import TASK_CONFIGS
from ..types import RaceResult, ScoreCard, TaskSpec


# Map task_id -> race MCP server module string.
_TASK_TO_SERVER = {
    "summarize_repo": "a2a_vs_mcp.mcp_servers.race_github",
    "negotiate_meeting": "a2a_vs_mcp.mcp_servers.race_calendar",
    "book_travel": "a2a_vs_mcp.mcp_servers.race_travel",
}


def _arm_faults(failure_script: list[FailureScriptEntry]) -> dict[str, ActiveFault]:
    """Build ACTIVE_FAULTS dict from failure_script (one fault armed per target)."""
    armed: dict[str, ActiveFault] = {}
    for entry in failure_script:
        armed[entry.target] = ActiveFault(
            fault_id=f"fault-{uuid.uuid4().hex[:8]}",
            kind=entry.kind,
            target=entry.target,
        )
    return armed


def _detect_and_record(events_before: int, recorder: TraceRecorder, detectors: list[Detector], lane: str) -> int:
    """Scan events appended since events_before. Returns new event count."""
    events = recorder.events
    for event in events[events_before:]:
        if event.get("event_type") == "fault_injected":
            detectors.append(Detector(
                fault_id=event["fault_id"],
                fault_kind=event.get("fault_kind", ""),
                target=event.get("target", ""),
                fault_inject_turn=event.get("turn_index", 0),
            ))
        for d in list(detectors):
            if d.state.value == "observed":
                continue
            if d.consume(event):
                wasted = compute_wasted_tokens(events, d.fault_id, lane)
                recorder.record(
                    "fault_observed",
                    fault_id=d.fault_id,
                    fault_kind=d.fault_kind,
                    target=d.target,
                    t_observed_ms=d.t_observed_ms or int(time.time() * 1000),
                    evidence=d.evidence_kind or "",
                    wasted_tokens_before_detection=wasted,
                )
    return len(events)


async def run_pure_mcp(
    task_spec: TaskSpec,
    run_id: str,
    recorder: TraceRecorder,
    failure_script: list[FailureScriptEntry],
    sonnet_client: Any = None,
) -> RaceResult:
    """RACE-02 pure_mcp lane runner."""
    lane = "pure_mcp"
    server_module = _TASK_TO_SERVER[task_spec.task_id]
    armed = _arm_faults(failure_script)
    fault_tok = set_active_faults(armed)
    ctx_tok = set_mcp_tool_context(recorder=recorder, run_id=run_id)
    detectors: list[Detector] = []
    events_before = len(recorder.events)
    started_at = time.time()
    score_pass = False
    try:
        client = MCPClient(server_module=server_module, trace=recorder, transport="in_process", failure_config=None)
        # v1 minimal: invoke each TARGETS function via tool name. The actual orchestration
        # logic (multi-turn LLM-driven tool selection) is the runner's job; for v1 the runner
        # iterates the task's TARGETS keys in failure_script order so faults always fire.
        cfg, targets, _ = TASK_CONFIGS[task_spec.task_id]
        results: dict[str, Any] = {}
        for entry in cfg.failure_script:
            tool_name = entry.target.split(".")[-1]
            try:
                args = _default_args_for_tool(task_spec.task_id, tool_name)
                results[tool_name] = client.call(tool_name, args)
            except InjectedFaultError as exc:
                recorder.record("tool_call", tool_name=tool_name, status="error", error_kind=str(exc))
            events_before = _detect_and_record(events_before, recorder, detectors, lane)
        # Synthesize task result for scorer.
        result = _result_from_outputs(task_spec.task_id, results)
        from importlib import import_module
        scorer = import_module(f"a2a_vs_mcp.race.tasks.{task_spec.task_id}").score
        sc: ScoreCard = scorer(result, recorder.events, judge=None)
        score_pass = sc.success
    finally:
        # Finalize detectors at done time.
        for d in detectors:
            d.finalize_at_done(score_pass)
        recorder.record("done", score_pass=score_pass, lane=lane, task_id=task_spec.task_id)
        events_before = _detect_and_record(events_before, recorder, detectors, lane)
        ACTIVE_FAULTS.reset(fault_tok)
        MCP_TOOL_CONTEXT.reset(ctx_tok)
    latency_ms = int((time.time() - started_at) * 1000)
    return RaceResult(
        run_id=run_id, lane=lane, task_id=task_spec.task_id,
        hardness_profile=task_spec.hardness_profile,
        score_card=ScoreCard(
            success=sc.success, ttff_ms=sc.ttff_ms, recovered=sc.recovered,
            wasted_tokens_before_detection=sc.wasted_tokens_before_detection,
            failure_mode=sc.failure_mode, cost_usd=sc.cost_usd, latency_ms=latency_ms,
        ),
        trace_id=run_id,
    )


def _default_args_for_tool(task_id: str, tool_name: str) -> dict[str, Any]:
    """v1 hardcoded args per (task, tool). Plan 11 tests assert these."""
    if task_id == "summarize_repo":
        if tool_name == "get_repo_metadata":

        if tool_name == "book_itinerary":

def _result_from_outputs(task_id: str, outputs: dict[str, Any]) -> dict[str, Any]:
    """Pack tool outputs into a scorer-friendly result dict."""
    if task_id == "summarize_repo":
        meta = outputs.get("get_repo_metadata") or {}
        files = outputs.get("list_files") or []
        modules = meta.get("modules", [])
        summary = f"{meta.get('purpose', '')} Modules include {', '.join(modules[:3])}. Entry point: {meta.get('entry_point', '')}."
        return {"summary": summary, "metadata": meta, "files": files}
    if task_id == "negotiate_meeting":

        return {"proposed_time": proposal, "free_busy_by_owner": {}}
    if task_id == "book_travel":
        booking = outputs.get("book_itinerary") or {}
        return {"booking": booking, "budget_usd": 1500, "summary": "business trip"}
    return {}
```

```
@dataclass
class FixtureBackedAgentHandler:
    """A2A agent handler that delegates to race.mocks (single chokepoint)."""
    capability: str
    recorder: TraceRecorder
    run_id: str
    task_id: str

    def handle_task(self, message: A2AMessage) -> AgentResult:
        key = (self.task_id, self.capability)
        if key not in _HANDLER_TABLE:
            return AgentResult(
                agent_id=f"race_{self.task_id}_agent",
                summary=f"unknown capability {self.capability}",
                details={}, confidence=0.0, status="failed",
            )
        fn_name, mock_mod = _HANDLER_TABLE[key]
        fn = getattr(mock_mod, fn_name)
        # Pass payload args + recorder/run_id kwargs.
        payload = dict(message.payload or {})
        try:
            data = fn(**payload, recorder=self.recorder, run_id=self.run_id)
            return AgentResult(
                agent_id=f"race_{self.task_id}_agent",
                summary=f"{self.capability} ok",
                details={"data": data}, confidence=1.0, status="completed",
            )
        except InjectedFaultError as exc:
            return AgentResult(
                agent_id=f"race_{self.task_id}_agent",
                summary=f"{self.capability} fault: {exc}",
                details={"error_kind": "injected"}, confidence=0.0, status="failed",
            )

def _detect_and_record(events_before: int, recorder: TraceRecorder, detectors: list[Detector], lane: str) -> int:
    events = recorder.events
```

```
async def _dispatch_step(step: dict, ctx: ExecutionContext, targets: dict, binds: dict, recorder: TraceRecorder) -> Any:
    """Execute a single hybrid_plan step; bind args via BINDS callables; call TARGETS callable."""
    target_name = step.get("target")
    bind_key = step.get("bind")
    args = binds[bind_key](ctx) if bind_key else step.get("args", {})
    fn = targets[target_name]
    recorder.record("tool_call", tool_name=target_name, status="start")
    result = fn(**args, recorder=recorder, run_id=ctx.get("run_id"))
    recorder.record("tool_call", tool_name=target_name, status="ok")
    return result


async def _execute_step_with_on_fault(step: dict, ctx: ExecutionContext, targets: dict, binds: dict, recorder: TraceRecorder) -> Any:
    """Execute a step with on_fault dispatch per D-29 enum."""
    on_fault = step.get("on_fault", "abort")
    try:
        return await _dispatch_step(step, ctx, targets, binds, recorder)
    except InjectedFaultError:
        if on_fault == "retry_once":
            try:
                return await _dispatch_step(step, ctx, targets, binds, recorder)
            except InjectedFaultError:
                raise
        elif on_fault == "delegate":
            alt = dict(step)
            alt["target"] = step.get("delegate_to") or step["target"]
            return await _dispatch_step(alt, ctx, targets, binds, recorder)
        elif on_fault == "abort":
            raise
        elif on_fault == "continue":
            return None
        raise
```
